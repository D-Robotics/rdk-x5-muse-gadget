cmake_minimum_required(VERSION 3.16)
project(muse_pet_camera_throttle)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
add_executable(camera_throttle src/camera_throttle.cpp)
target_compile_features(camera_throttle PRIVATE cxx_std_17)
target_compile_options(camera_throttle PRIVATE -Wall -Wextra -Wpedantic)
ament_target_dependencies(camera_throttle rclcpp sensor_msgs)
install(TARGETS camera_throttle DESTINATION lib/${PROJECT_NAME})
ament_package()

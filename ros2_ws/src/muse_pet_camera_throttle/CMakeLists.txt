cmake_minimum_required(VERSION 3.16)
project(muse_pet_camera_throttle)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(hbm_img_msgs REQUIRED)
add_executable(camera_throttle src/camera_throttle.cpp)
target_compile_features(camera_throttle PRIVATE cxx_std_17)
target_compile_options(camera_throttle PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(camera_throttle PRIVATE include)
ament_target_dependencies(camera_throttle rclcpp sensor_msgs std_msgs hbm_img_msgs)
add_executable(depth_input src/depth_input.cpp)
target_compile_features(depth_input PRIVATE cxx_std_17)
target_compile_options(depth_input PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(depth_input PRIVATE include)
ament_target_dependencies(depth_input rclcpp sensor_msgs hbm_img_msgs)
install(TARGETS camera_throttle depth_input DESTINATION lib/${PROJECT_NAME})
include(CTest)
if(BUILD_TESTING)
  add_executable(stream_budget_test test/test_stream_budget.cpp)
  target_include_directories(stream_budget_test PRIVATE include)
  target_compile_features(stream_budget_test PRIVATE cxx_std_17)
  add_test(NAME stream_budget COMMAND stream_budget_test)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_executable(shared_transport_probe test/shared_transport_probe.cpp)
  target_compile_features(shared_transport_probe PRIVATE cxx_std_17)
  ament_target_dependencies(shared_transport_probe rclcpp sensor_msgs std_msgs hbm_img_msgs)
  add_test(NAME shared_transport
    COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test/test_shared_transport.py
      --relay $<TARGET_FILE:camera_throttle> --probe $<TARGET_FILE:shared_transport_probe>)
  set_tests_properties(shared_transport PROPERTIES TIMEOUT 35)
  add_test(NAME shared_transport_fast_preview
    COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test/test_shared_transport.py
      --relay $<TARGET_FILE:camera_throttle> --probe $<TARGET_FILE:shared_transport_probe> --fast-preview)
  set_tests_properties(shared_transport_fast_preview PROPERTIES TIMEOUT 35)
endif()
ament_package()
